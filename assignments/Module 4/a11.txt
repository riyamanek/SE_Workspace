5.Add a theme switcher button to your page that toggles between a light and dark background using Bootstrap classes, and updates the text color for readability.<br><br><em><strong>Hint:</strong> Use JavaScript to toggle 'bg-light'/'bg-dark' and 'text-dark'/'text-light' classes on the main container.</em>

<!DOCTYPE html>
<html>
<head>
    <title>Theme Switcher</title>
    <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.8/dist/css/bootstrap.min.css" rel="stylesheet">
</head>
<body>
<div id="main" class="container bg-light text-dark p-5">
    <h2>Movie Booking</h2>
    <p>Welcome to BookMyShow</p>
    <button onclick="changeTheme()" class="btn btn-primary">
        Change Theme
    </button>
</div>
<script>
function changeTheme() {
    let main = document.getElementById("main");
    main.classList.toggle("bg-light");
    main.classList.toggle("bg-dark");
    main.classList.toggle("text-dark");
    main.classList.toggle("text-light");

}
</script>
</body>
</html>
