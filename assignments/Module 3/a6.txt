1. Create a class called Song in your preferred OOP language with private properties title and artist. Add public getter and setter methods to access and modify these properties, then create an object and update its title.

#include <iostream>
using namespace std;

class Song
{
  private:
  string title;
  string artist;
  
  public:
  
  void setter(string t,string a)
  {
      title=t;
      artist=a;
  }
  void getter()
  {
      cout<<"Title:-"<<title<<endl;
      cout<<"Artist:-"<<artist<<endl;
  }
  
};
int main()
{
    Song s;
    
    cout<<"Before modify"<<endl;
    s.setter("perfect","Ed sheeran");
    s.getter();
    
    cout<<"After modify"<<endl;
    s.setter("Shape of you","Ed sheeran");
    s.getter();

    return 0;
}
