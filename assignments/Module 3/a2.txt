using namespace std;

class Playlist
{
    public:
    bool isPublic;
    
    void togglePublic()
    {
        isPublic=!isPublic;
    }
};
int main(){
    
    Playlist p;
    p.isPublic=true;
    cout<<"Before use togglePublic function:-"<<p.isPublic<<endl;
    
    p.togglePublic();
    cout<<"After use togglePublic function:-"<<p.isPublic<<endl;
    
    p.togglePublic();
    cout<<"After once again call togglePublic function:-"<<p.isPublic<<endl;
    
    return 0;
}

4.Extend your Playlist class to include a member function addSong(songTitle) that adds the song title to an array property called songs. Demonstrate by adding three song titles and displaying the updated songs list.<br><br><em><strong>Hint:</strong> Initialize songs as an empty array inside the constructor.</em>

#include <iostream>
using namespace std;

class Playlist
{
public:
    string name;
    string songs[10];
    int count;

    Playlist()
    {
        name = "My Playlist";
        count = 0;
    }

    void addSong(string songTitle)
    {
        songs[count] = songTitle;
        count++;
    }

    void display()
    {
        cout << "Playlist Name: " << name << endl;

        for(int i = 0; i < count; i++)
        {
            cout << songs[i] << endl;
        }
    }
};

int main()
{
    Playlist p;

    p.addSong("Perfect");
    p.addSong("Memories");
    p.addSong("Untill I found you");
    p.addSong("Attention");

    p.display();

    return 0;
}
