    function parseProfileUrl(value, allowedHosts) {
      let url;
      try {
        url = new URL(value);
      } catch {
        return null;
      }

      const hostname = url.hostname.toLowerCase().replace(/^www\./, "");
      const isAllowedHost = allowedHosts.some((host) => hostname === host || hostname.endsWith(`.${host}`));
      if (!isAllowedHost || !["http:", "https:"].includes(url.protocol) || url.username || url.password) return null;
      return url;
    }

    function getHandle(url) {
      const segments = url.pathname.split("/").filter(Boolean);
      const ignoredSegments = new Set(["user", "artist", "channel", "c", "profile", "in", "intl-en", "us"]);
      const handle = [...segments].reverse().find((segment) => !ignoredSegments.has(segment.toLowerCase()));
      if (!handle) return url.hostname.replace(/^www\./, "");
      try {
        return `@${decodeURIComponent(handle).replace(/^@/, "")}`;
      } catch {
        return `@${handle.replace(/^@/, "")}`;
      }
    }

      logo.className = "preview-logo";

      info.className = "preview-info";

      title.className = "preview-platform";
      title.textContent = platform.name;
      const handle = document.createElement("span");
      handle.className = "preview-handle";
      handle.textContent = getHandle(url);

      const entries = platforms
        .map((platform) => ({
          platform,
          value: form.elements[platform.input].value.trim()
        }))
        .filter(({ value }) => value !== "");

      if (entries.length === 0) {
        formMessage.textContent = "Add at least one Instagram, LinkedIn, or Spotify URL to create your profile.";
        return;
      }

      const invalid = entries.find(({ platform, value }) => !parseProfileUrl(value, platform.hosts));
      if (invalid) {
        formMessage.textContent = `Enter a valid ${invalid.platform.name} profile URL (including https://).`;
        form.elements[invalid.platform.input].focus();
        return;
      }

      const photoInput = form.elements.profilePhoto;
      const photo = photoInput.files[0];
      if (photo && !photo.type.startsWith("image/")) {
        formMessage.textContent = "Choose an image file for your profile picture.";
        photoInput.focus();
        return;
      }
      if (photo && photo.size > 5 * 1024 * 1024) {
        formMessage.textContent = "Your profile picture must be 5 MB or smaller.";
        photoInput.focus();
        return;
      }

      const name = form.elements.displayName.value.trim();
      const username = form.elements.username.value.trim().replace(/^@/, "");
      const bio = form.elements.bio.value.trim();
      const fragment = document.createDocumentFragment();
      const profileFragment = document.createDocumentFragment();
      for (const { platform, value } of entries) {
        const url = parseProfileUrl(value, platform.hosts);
        fragment.append(makePreviewCard(platform, url));
        profileFragment.append(makeProfileLink(platform, url));
      }
      previewList.replaceChildren(fragment);
      profileLinks.replaceChildren(profileFragment);
      cardCount.textContent = `${entries.length} ${entries.length === 1 ? "link" : "links"}`;
      profileName.textContent = name;
      profileHandle.textContent = username ? `@${username}` : "";
      profileBio.textContent = bio || "Find me around the web.";
      profileAvatar.setAttribute("aria-label", `${name}'s profile picture`);
      if (photo) {
        if (avatarObjectUrl) URL.revokeObjectURL(avatarObjectUrl);
        avatarObjectUrl = URL.createObjectURL(photo);
        const image = document.createElement("img");
        image.src = avatarObjectUrl;
        image.alt = "";
        profileAvatar.replaceChildren(image);
      } else if (!avatarObjectUrl) {
        profileAvatar.textContent = name.charAt(0).toUpperCase();
      }
      document.title = `${name} — Bio Link Page`;
      formMessage.textContent = "Your profile page is ready in the preview above.";
      formMessage.classList.add("success");
      document.querySelector(".profile-card").scrollIntoView({ behavior: "smooth", block: "start" });
    });
  </script>
