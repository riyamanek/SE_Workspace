{

    printf("enter your size of array");
    scanf("%d",&n);
    
    printf("enter your elements");
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
        
    }
    int highest=arr[0];
    for(i=1;i<n;i++)

        if(arr[i]>highest)
        {
            highest=arr[i];
        }
    }
    printf("Highest digit is %d",highest);

}

3)From Given Array Find sum of all Digit

#include <stdio.h>
#include <conio.h>
void main()
{
  int i,n;
  int arr[n];
  printf("enter size of array");
  scanf("%d",&n);
  for(i=0;i<n;i++)
  {
      scanf("%d",&arr[i]);
  }
   int sum=0;
  for(i=0;i<n;i++)
  {
      int num=arr[i];
     while(num>0)
     {
         sum+=num%10;
         num=num/10;
     }
  }
    printf(" sum of all digits%d",sum);
   getch();
}
