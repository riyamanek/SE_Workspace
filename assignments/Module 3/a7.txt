5.Given a file called insta_followers.txt containing Instagram usernames (one per line), write a program to count and display the total number of followers listed in the file.<br><br><em><strong>Constraint:</strong> Do not use any array or vector to store the names—just count as you read.</em>

#include <iostream>
#include <fstream>
using namespace std;

int main()
{
    string username;
    ofstream outfile;
    outfile.open("insta_followers.txt");
    
    cout<<"Add your instagram username"<<endl;
    
    for(int i=0;i<5;i++)
    {
        cout<<"enter your username:-";
        getline(cin,username);
        outfile<<username<<endl;
    }
    
    outfile.close();
    
    int count=0;
    ifstream infile;
    infile.open("insta_followers.txt");
    while(getline(infile,username))
    {
        cout<<username<<endl;
        count++;
    }
    cout<<"Total username:-"<<count<<endl;
    
    infile.close();
    
    return 0;
    
}
