1.Create a Java class called PaymentProcessor with two overloaded methods processPayment(): one that takes only an amount, and one that takes amount and a coupon code. Print which version is called and the final amount in each case.


#include <iostream>
using namespace std;

class PaymentProcessor
{
    public:
    void processPayment(double amount)
    {
       cout<<"Amount:-"<<amount<<endl;
    }
    void processPayment(double amount,string coupon_code)
    {
        cout<<"Amount:-"<<amount<<" "<<"Coupon code:-"<<coupon_code<<endl;
        
        if(coupon_code =="cp100")
        {
            amount=amount-(amount*0.10);
        }
        cout<<"Final Amount:-"<<amount<<endl;
    }
};
int main()
{
   PaymentProcessor p;
   p.processPayment(1000);
   p.processPayment(1000,"cp100");

    return 0;
}
