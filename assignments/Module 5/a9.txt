4.Write a trigger on a Wallet table that automatically deducts the purchase amount from the user's balance whenever a new transaction is inserted, similar to how Paytm updates wallet balance after a payment.

CREATE TABLE Wallet (
    user_id INT PRIMARY KEY,
    balance DECIMAL(10,2)
);
INSERT INTO Wallet VALUES (101, 5000.00);

CREATE TABLE Transactions (
    transaction_id INT PRIMARY KEY,
    user_id INT,
    purchase_amount DECIMAL(10,2)
);

DELIMITER //
CREATE TRIGGER deduct_balance
AFTER INSERT ON Transactions FOR EACH ROW
BEGIN
update wallet set balance=balance-new.purchase_amount
where user_id=new.user_id;
END //
DELIMITER ;

INSERT INTO Transactions
VALUES (1, 101, 1200.00);

select*from wallet
