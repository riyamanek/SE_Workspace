2.Create a PL/SQL block that uses an IF statement to check if a given order amount is above 500. If yes, print 'Eligible for free delivery', else print 'Delivery charges apply'.

DELIMITER //
CREATE PROCEDURE DELIVERY()
BEGIN
DECLARE ORDER_AMOUNT INT;
SET ORDER_AMOUNT=1000;
IF ORDER_AMOUNT>500 THEN
	SELECT 'eligible for free delivery' AS Message;
 else
	SELECT 'DELIVERY CHARGES' AS Message;
end if;    
END //
DELIMITER ;
CALL DELIVERY();

3.Write a PL/SQL block that uses a simple LOOP to print the numbers 1 to 5 using DBMS_OUTPUT.PUT_LINE.<br><br><em><strong>Hint:</strong> Use a counter variable and EXIT WHEN condition inside the loop.</em>

DELIMITER //
CREATE PROCEDURE PRINT_LOOP()
BEGIN
    DECLARE COUNTER INT DEFAULT 1;
    NUM_LOOP:LOOP
    SELECT COUNTER;
    SET COUNTER=COUNTER+1;
    IF COUNTER>5 THEN
    LEAVE NUM_LOOP;
    END IF;
    END LOOP NUM_LOOP;
END //
DELIMITER ;
call PRINT_LOOP();


4.Declare a cursor in PL/SQL to select all product names from a table called PRODUCTS. Fetch each product name and print it using DBMS_OUTPUT.PUT_LINE.<br><br><em><strong>Hint:</strong> Assume the PRODUCTS table has a column PRODUCT_NAME.</em>

CREATE TABLE PRODUCTS (
    PRODUCT_NAME VARCHAR(50)
);
INSERT INTO PRODUCTS VALUES("lAPTOP"),("CHARGER"),("MOBILE")

DELIMITER //
CREATE PROCEDURE SHOW_PRODUCTS12()
BEGIN
DECLARE DONE INT DEFAULT 0;
 DECLARE P_NAME VARCHAR(50);
  DECLARE SHOW_PRODUCT CURSOR FOR
   SELECT PRODUCT_NAME
    FROM PRODUCTS;
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET DONE = 1;
    OPEN SHOW_PRODUCT;
    product_loop: LOOP
    FETCH SHOW_PRODUCT INTO P_NAME;
   IF DONE = 1 THEN
   LEAVE product_loop;
  END IF;
 SELECT P_NAME AS PRODUCT_NAME;
 END LOOP product_loop;
 Close Cursor
CLOSE SHOW_PRODUCT;
END //
DELIMITER ;
CALL SHOW_PRODUCTS12();
