1.Create a MySQL table called playlists to store Spotify-style music playlists, with columns: playlist_id (primary key), user_id (foreign key), name (varchar), and created_at (datetime). Write the CREATE TABLE statement.

create table SpotifyUser (
    user_id INT PRIMARY KEY,
    username VARCHAR(50) UNIQUE,
    email VARCHAR(100) NOT NULL,
    subscription_type VARCHAR(20)
);
create table playlists(
playlist_id int primary key,
user_id int,
name varchar(50),
created_at datetime,
foregign key (user_id)references SpotifyUser(user_id)
);

5.Create a stored procedure named GetMonthlyPlaylistCount that takes a user_id and a month (as an integer) as input, and returns the total number of playlists the user created in that month.<br><br><em><strong>Hint:</strong> Use the MONTH() function in your WHERE clause.</em>

DELIMITER //
create PROCEDURE GetMonthlyPlaylistCount
(IN p_user_id int,IN p_month int)
begin
select count(*)AS total_playlists from playlists
where user_id=p_user_id AND month(created_at)=p_month;
end //
